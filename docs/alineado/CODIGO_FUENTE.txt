     7	/**
     8	 * Auto-alineado: al convertir un gesto, ajusta el rectángulo donde va la salida para que quede en
     9	 * el mismo renglón que las formas ya reconocidas.
    10	 *
    11	 * - Un cierre de paréntesis, corchete o llave toma la altura y la posición vertical de su apertura:
    12	 *   la última sin cerrar de su misma expresión (formas encadenadas a menos de [MAX_GAP] altos; la
    13	 *   escritura a mano del renglón también encadena), en su línea (a menos de [BRACKET_OFFSET] del alto)
    14	 *   y no más de [MAX_RESIZE] veces más alta o más baja que él.
    15	 * - Una apertura (o un cierre sin pareja) se centra en lo que tiene al lado (no encima ni debajo: a lo
    16	 *   sumo [BESIDE_OVERLAP] de solape), en su línea y a menos de [MAX_GAP] altos; junto a una fracción,
    17	 *   en su barra; junto a una integral, en su medio. Solo toma la altura de otro signo de agrupación
    18	 *   parecido.
    19	 * - Un número o una letra toma la escala (el alto sobre la línea) de la letra o el número más cercano
    20	 *   del renglón y se apoya en su piso; solo si está a menos de [MAX_GAP] altos, así no toma el de otra
    21	 *   cuenta del otro lado de la hoja. Un número prefiere otro número a su alcance. Las integrales,
    22	 *   raíces y sumatorias solo se centran en su línea.
    23	 * - Los "contenedores" (paréntesis, integrales, raíz, fracción, sumatoria) no sirven de referencia
    24	 *   para las demás formas: un número dentro de una matriz no se centra en el corchete.
    25	 * - Los signos de operación (+, −, ×, ÷, =, <…) tampoco: no marcan la altura del renglón. Ellos se
    26	 *   centran en la línea de las letras y números más cercanos, sin cambiar de tamaño.
    27	 * - Lo escrito claramente más arriba o más abajo que la referencia (más de [OFFSET] de su alto: un
    28	 *   exponente, un numerador, un denominador) queda donde se escribió. Una raya con algo justo encima
    29	 *   (la barra de una fracción) tampoco se mueve.
    30	 * - Un exponente o subíndice (más chico que la forma de al lado y corrido hacia arriba o abajo de su
    31	 *   renglón) queda donde se escribió, y uno ya escrito no sirve de referencia: lo que sigue a x² se
    32	 *   alinea con la x, no con el 2. Un signo arriba del renglón (el − de x⁻¹) tampoco baja.
    33	 * - Una forma lejos de todas (más de [MAX_GAP] altos de separación) no se alinea con nada.
    34	 *
    35	 * Solo mira la tinta que salió de un patrón; la escritura a mano nunca es referencia, solo une la
    36	 * expresión de un paréntesis ([bridges]).
    37	 */
    38	object AutoAlign {

   100	    /**
   101	     * Como el otro [anchor], con los trazos escritos a mano de la página ([bridges]): un cierre los usa
   102	     * para encadenar su expresión hasta la apertura, pero nunca como referencia.
   103	     *
   104	     * Si el patrón tiene piso elegido a mano y no hay con qué alinear, lo escrito es la parte de arriba
   105	     * del piso: la salida apoya su piso en el borde de abajo del trazo (el "lim" de "lim→" queda en la
   106	     * línea y la flecha debajo), en vez de achicarse entera dentro del trazo.
   107	     */
   108	    fun anchor(
   109	        pattern: InkPattern,
   110	        gesture: Bounds,
   111	        items: List<Item>,
   112	        bridges: List<Bounds>,
   113	        lookup: (String) -> InkPattern?,
   114	    ): Bounds {
   115	        // Con núcleo, lo escrito ya es el núcleo: la salida lo reemplaza en el mismo lugar y tamaño.
   116	        if (PatternCore.isPartial(pattern)) return gesture
   117	        val out = rowAnchor(pattern, gesture, items, bridges, lookup)
   118	        if (out !== gesture || pattern.floor == null || pattern.stretch != null || gesture.height <= 0.5f) return out
   119	        val scale = pattern.outputScale.coerceAtLeast(0.01f)
   120	        val f = floor(pattern)
   121	        val h = gesture.height * scale / f
   122	        val ah = h / scale
   123	        val aw = gesture.width * ah / gesture.height
   124	        val cy = gesture.bottom - h * f + h / 2f
   125	        return Bounds(gesture.centerX - aw / 2f, cy - ah / 2f, gesture.centerX + aw / 2f, cy + ah / 2f)
   126	    }

   128	    private fun rowAnchor(
   129	        pattern: InkPattern,
   130	        gesture: Bounds,
   131	        items: List<Item>,
   132	        bridges: List<Bounds>,
   133	        lookup: (String) -> InkPattern?,
   134	    ): Bounds {
   135	        if (pattern.stretch != null || gesture.height <= 0.5f || items.isEmpty()) return gesture
   136	        val row = items.filter { sameRow(it.bounds, gesture) }
   137	        if (row.isEmpty()) return gesture
   138	        val scale = pattern.outputScale.coerceAtLeast(0.01f)
   139	
   140	        if (gesture.height < gesture.width * 0.4f && items.any { stacked(it.bounds, gesture) }) return gesture
   141	        kindOf(pattern)?.let { kind ->
   142	            return bracketAnchor(kind, gesture, items, row, bridges.filter { sameRow(it, gesture) }, lookup, scale)
   143	        }
   144	        val sign = isSign(pattern)
   145	        val variable = isVariable(pattern)
   146	        // Los exponentes y subíndices ya escritos no marcan el renglón: lo que sigue a x² se alinea con la x.
   147	        val bases = row.filter { r -> lookup(r.patternId)?.let { !isContainer(it) && !isSign(it) } ?: true }
   148	        val plain = row.filter { r -> bases.none { b -> b !== r && isScript(r.bounds, r.bounds.height, b.bounds) } }
   149	        val predicted = gesture.height * scale
   150	        fun near(r: Item) = horizontalGap(r.bounds, gesture) <= MAX_GAP * max(r.bounds.height, predicted)
   151	        // Un número prefiere otro número a su alcance; si no hay, toma la letra o el número más cercano.
   152	        val refs = if (isContainer(pattern)) plain else plain.filter { r -> lookup(r.patternId)?.let { !isContainer(it) && !isSign(it) } ?: true }
   153	            .let { all -> if (variable) all else all.filter { r -> near(r) && lookup(r.patternId)?.let(::isVariable) != true }.ifEmpty { all } }
   154	        // Un signo sin letras ni números al lado se centra en otro signo; lo demás nunca toma su altura.
   155	        val ref = (refs.takeIf { it.isNotEmpty() || !sign } ?: plain.filter { r -> lookup(r.patternId)?.let(::isSign) == true })
   156	            .minByOrNull { horizontalGap(it.bounds, gesture) } ?: return gesture
   157	        // Lejos de todo (otra cuenta en la misma altura de la hoja): queda donde se escribió.
   158	        if (horizontalGap(ref.bounds, gesture) > MAX_GAP * max(ref.bounds.height, predicted)) return gesture
   159	        if (ref in refs) {
   160	            val out = Bounds(gesture.left, gesture.centerY - predicted / 2f, gesture.right, gesture.centerY + predicted / 2f)
   161	            if (sign) {
   162	                // Un signo claramente arriba (x⁻¹) o debajo del renglón se escribió así a propósito.
   163	                val h = ref.bounds.height
   164	                if (out.centerY < ref.bounds.top + SIGN_RAISED * h || out.centerY > ref.bounds.bottom + SIGN_LOWERED * h) return gesture
   165	            } else if (isScript(out, predicted, ref.bounds)) {
   166	                return gesture
   167	            }
   168	        }
   169	        val refPattern = lookup(ref.patternId)
   170	        // Números y letras se apoyan en el piso de la forma de al lado y toman su escala.
   171	        val byFloor = ref in refs && (pattern.floor != null || (!sign && !isContainer(pattern)))
   172	        // Entre formas de distinto alto, el desvío se mide con la más alta de las dos.
   173	        val reach = if (sign) ref.bounds.height else max(ref.bounds.height, predicted)
   174	        if (ref in refs && abs(ref.bounds.centerY - gesture.centerY) > OFFSET * reach) return gesture
   175	        if (byFloor) {
   176	            // La salida se dibuja centrada en el anclaje y mide su alto × scale: su piso queda a floor(pattern) de su arriba.
   177	            val refFrac = refPattern?.let(::floor) ?: 1f
   178	            val refFloor = ref.bounds.top + ref.bounds.height * refFrac
   179	            val f = floor(pattern)
   180	            // Misma escala que la de al lado: igual cuerpo sobre la línea (las colas de g, y… no cuentan).
   181	            val h = if (refPattern != null && isGlyph(refPattern) && isGlyph(pattern)) ref.bounds.height * refFrac / f else predicted
   182	            val ah = h / scale
   183	            val aw = gesture.width * ah / gesture.height
   184	            val cy = refFloor - h * f + h / 2f
   185	            return Bounds(gesture.centerX - aw / 2f, cy - ah / 2f, gesture.centerX + aw / 2f, cy + ah / 2f)
   186	        }
   187	        val dy = ref.bounds.centerY - gesture.centerY
   188	        return Bounds(gesture.left, gesture.top + dy, gesture.right, gesture.bottom + dy)
   189	    }

   191	    /** Letra o número (no signo, contenedor, coma, %, ! ni barra): lo que iguala su escala con el renglón. */
   192	    private fun isGlyph(p: InkPattern): Boolean {
   193	        if (p.stretch != null || isSign(p) || isContainer(p)) return false
   194	        val key = p.id.removePrefix(PresetPatterns.PREFIX)
   195	        return !(p.builtIn && (key in TALL_OPERATORS || key in BASELINE_MARKS))
   196	    }

   198	    /** Paréntesis, corchetes y llaves (ver el KDoc del objeto). */
   199	    private fun bracketAnchor(
   200	        kind: String, gesture: Bounds, items: List<Item>, row: List<Item>, bridges: List<Bounds>,
   201	        lookup: (String) -> InkPattern?, scale: Float,
   202	    ): Bounds {
   203	        val predicted = gesture.height * scale
   204	        // 1. Un cierre copia su apertura (misma expresión, en su línea, tamaño a lo sumo MAX_RESIZE veces distinto).
   205	        partnerFor(kind, gesture, row, bridges, predicted, lookup)?.let { return resized(gesture, it.bounds, scale) }
   206	        // 2. Si no, se centra en lo que tiene al lado; los exponentes y subíndices no cuentan.
   207	        val bases = row.filter { r -> lookup(r.patternId)?.let { !isContainer(it) && !isSign(it) } ?: true }
   208	        val plain = row.filter { r -> bases.none { b -> b !== r && isScript(r.bounds, r.bounds.height, b.bounds) } }
   209	        // Junto a una fracción, la referencia es su barra (si está en la línea del paréntesis).
   210	        val bars = fractionBars(items, lookup)
   211	        val ref = plain
   212	            .map { r -> barOf(r, bars)?.takeIf { onLine(it.bounds, gesture, predicted) } ?: r }
   213	            .distinct()
   214	            .filter {
   215	                onLine(it.bounds, gesture, predicted) && beside(it.bounds, gesture) &&
   216	                    horizontalGap(it.bounds, gesture) <= MAX_GAP * max(it.bounds.height, predicted)
   217	            }
   218	            .minWithOrNull(compareBy<Item>({ horizontalGap(it.bounds, gesture) }, { abs(it.bounds.centerY - gesture.centerY) }))
   219	            ?: return gesture
   220	        // 3. Solo copia la altura de otro signo de agrupación parecido.
   221	        if (lookup(ref.patternId)?.let(::kindOf) != null) {
   222	            val ratio = predicted / ref.bounds.height.coerceAtLeast(0.5f)
   223	            if (ratio in (1f / SIMILAR_RATIO)..SIMILAR_RATIO) return resized(gesture, ref.bounds, scale)
   224	        }
   225	        // Junto a un número, una letra, un signo o una integral se centra en su medio, sin cambiar de tamaño.
   226	        val dy = ref.bounds.centerY - gesture.centerY
   227	        return Bounds(gesture.left, gesture.top + dy, gesture.right, gesture.bottom + dy)
   228	    }

   230	    /** Apertura de [kind] si es un cierre: la última sin pareja de su expresión, en su línea y de tamaño parecido. */
   231	    private fun partnerFor(
   232	        kind: String, gesture: Bounds, row: List<Item>, bridges: List<Bounds>, predicted: Float,
   233	        lookup: (String) -> InkPattern?,
   234	    ): Item? {
   235	        // Las aperturas no buscan pareja a su derecha.
   236	        val open = PAIRS.entries.firstOrNull { it.value == kind }?.key ?: return null
   237	        val stack = ArrayDeque<Item>()
   238	        val left = sameExpression(gesture, row, bridges, predicted)
   239	            .filter { it.bounds.centerX < gesture.centerX && onLine(it.bounds, gesture, predicted) }
   240	            .sortedBy { it.bounds.centerX }
   241	        for (it in left) {
   242	            when (lookup(it.patternId)?.let(::kindOf)) {
   243	                open -> stack.addLast(it)
   244	                kind -> stack.removeLastOrNull()
   245	            }
   246	        }
   247	        val top = stack.lastOrNull() ?: return null
   248	        val ratio = predicted / top.bounds.height.coerceAtLeast(0.5f)
   249	        return top.takeIf { ratio in (1f / MAX_RESIZE)..MAX_RESIZE }
   250	    }
   251	
   252	    /**
   253	     * Lo del renglón encadenado con el gesto: cada forma a menos de [MAX_GAP] altos de lo ya juntado.
   254	     * Los trazos a mano del renglón ([bridges]) alargan la cadena (x + y escritos sin convertir entre
   255	     * los dos paréntesis), pero nunca se devuelven: no son referencia de nada.
   256	     */
   257	    private fun sameExpression(gesture: Bounds, row: List<Item>, bridges: List<Bounds>, predicted: Float): List<Item> {
   258	        val rest = row.toMutableList()
   259	        val links = bridges.toMutableList()
   260	        val inside = ArrayList<Item>()
   261	        var span = gesture
   262	        while (true) {
   263	            val next = rest.firstOrNull { horizontalGap(it.bounds, span) <= MAX_GAP * max(it.bounds.height, predicted) }
   264	            if (next != null) {
   265	                rest.remove(next); inside += next; span = span.union(next.bounds); continue
   266	            }
   267	            val link = links.firstOrNull { horizontalGap(it, span) <= MAX_GAP * max(it.height, predicted) } ?: break
   268	            links.remove(link); span = span.union(link)
   269	        }
   270	        return inside
   271	    }

   273	    /** [b] está en la línea del paréntesis: su centro a menos de [BRACKET_OFFSET] del mayor de los dos altos. */
   274	    private fun onLine(b: Bounds, gesture: Bounds, predicted: Float) =
   275	        abs(b.centerY - gesture.centerY) <= BRACKET_OFFSET * max(b.height, predicted)
   276	
   277	    /** [b] está al costado de [g], no encima ni debajo: se solapan en horizontal a lo sumo [BESIDE_OVERLAP] del más angosto. */
   278	    private fun beside(b: Bounds, g: Bounds): Boolean {
   279	        val overlap = min(b.right, g.right) - max(b.left, g.left)
   280	        return overlap <= BESIDE_OVERLAP * min(b.width, g.width)
   281	    }

   283	    private fun above(i: Item, bar: Item): Boolean {
   284	        val b = bar.bounds
   285	        val s = b.width * 0.1f
   286	        return i !== bar && i.bounds.centerX in (b.left - s)..(b.right + s) &&
   287	            i.bounds.bottom <= b.centerY && b.top - i.bounds.bottom <= 0.8f * i.bounds.height
   288	    }
   289	
   290	    private fun below(i: Item, bar: Item): Boolean {
   291	        val b = bar.bounds
   292	        val s = b.width * 0.1f
   293	        return i !== bar && i.bounds.centerX in (b.left - s)..(b.right + s) &&
   294	            i.bounds.top >= b.centerY && i.bounds.top - b.bottom <= 0.8f * i.bounds.height
   295	    }

   297	    /** Barras de fracción: rayas con algo justo encima y justo debajo (también la de un resultado, de alto 0). */
   298	    private fun fractionBars(items: List<Item>, lookup: (String) -> InkPattern?): List<Item> =
   299	        items.filter { isBar(it, lookup) }.filter { bar -> items.any { above(it, bar) } && items.any { below(it, bar) } }
   300	
   301	    /** La barra (la más ancha) de la fracción de la que [item] es numerador o denominador. */
   302	    private fun barOf(item: Item, bars: List<Item>): Item? =
   303	        bars.filter { above(item, it) || below(item, it) }.maxByOrNull { it.bounds.width }
   304	
   305	    private val BAR_IDS = setOf(DefaultPatterns.FRACTION, PresetPatterns.PREFIX + "menos", PresetPatterns.PREFIX + "barra")
   306	
   307	    private fun isBar(i: Item, lookup: (String) -> InkPattern?): Boolean {
   308	        if (i.bounds.height >= 0.4f * i.bounds.width) return false
   309	        if (i.patternId in BAR_IDS) return true
   310	        val p = lookup(i.patternId) ?: return false
   311	        return MathNotes.label(p.name) in MathNotes.BAR_LABELS
   312	    }

   314	    /** "parentesis-abre", "corchete-cierra"… o null si no es un signo de agrupación con pareja. */
   315	    private fun kindOf(p: InkPattern): String? {
   316	        val key = p.id.removePrefix(PresetPatterns.PREFIX)
   317	        if (key in PAIRS.keys || key in PAIRS.values) return key
   318	        return NAMES[fold(p.name)]
   319	    }
   320	
   321	    /**
   322	     * Signo de operación que va centrado en la línea (+, −, ×, ÷, =, <, ≈…): no marca la altura del
   323	     * renglón. Los de la biblioteca por su sección; los propios del usuario por su nombre.
   324	     */
   325	    fun isSign(p: InkPattern): Boolean {
   326	        if (p.stretch != null) return false
   327	        val key = p.id.removePrefix(PresetPatterns.PREFIX)
   328	        if (p.builtIn && p.id.startsWith(PresetPatterns.PREFIX)) return p.category == PatternCategory.OPERATORS && key !in TALL_OPERATORS && key !in BASELINE_MARKS
   329	        return isSignName(p.name)
   330	    }
   331	
   332	    /**
   333	     * Variable: una letra que se usa como incógnita (x, y, a…) o una letra de la biblioteca (griegas,
   334	     * letras y funciones). No cambia de tamaño al alinearse.
   335	     */
   336	    fun isVariable(p: InkPattern): Boolean {
   337	        if (p.stretch != null) return false
   338	        val u = p.unknown?.trim()
   339	        if (u != null) return u.isNotEmpty()
   340	        if (p.builtIn && p.id.removePrefix(PresetPatterns.PREFIX) in BASELINE_MARKS) return true
   341	        if (p.category == PatternCategory.GREEK || p.category == PatternCategory.LETTERS) return true
   342	        return letterOf(p) != null
   343	    }
   344

    27	     * Rectángulo para [InkReplacement.place] que deja [p] ocupando [rows] renglones, los más cercanos al

    31	        if (rows !in 1..MAX_ROWS || !exceedsRow(p)) return null
    32	        if (p.sizing == OutputSizing.FIXED || p.sizing == OutputSizing.MATCH_WIDTH) return null

    44	            if (core.height > 0.5f) core.width * h / core.height else gesture.width
    45	        }

   195	        val all = p.outputInk.indices.toList()
   196	        p.core?.filter { it in all }?.distinct()?.sorted()?.takeIf { it.isNotEmpty() }?.let { return it }
   197	        // Los símbolos de fábrica traen el núcleo marcado; sin marca son enteros.
   198	        if (p.builtIn || p.stretch != null || p.outputInk.size < 2) return all
   199	        val whole = p.outputInk.bounds()
   200	        if (whole.height <= 0.5f || whole.width <= 0.5f) return all
   201	        val bands = bands(p.outputInk).sortedByDescending { it.second.height }
   202	        if (bands.size < 2) return all
   203	        val (top, topBounds) = bands[0]
   204	        if (topBounds.height < 0.2f * whole.height || topBounds.height < BAND_LEAD * bands[1].second.height) return all
   205	        val gesture = p.gestureExamples.map { it.bounds() }.filter { it.height > 0.5f && it.width > 0.5f }
   206	        if (gesture.isEmpty()) return all
   207	        val target = gesture.map { ln((it.width / it.height).toDouble()) }.average()
   208	        fun dist(b: Bounds) = abs(ln((b.width.coerceAtLeast(0.5f) / b.height).toDouble()) - target)
   209	        return if (dist(topBounds) + MARGIN < dist(whole)) top else all
   210	    }

   226	                out[out.lastIndex] = last.first to last.second.union(b)
   227	            } else {
   228	                out += mutableListOf(i) to b
   229	            }
   230	        }

   250	     * @return trazos nuevos (ids nuevos, mismo [InkStroke.groupId], [InkStroke.patternId] = patrón).
   251	     */
   252	    fun place(
   253	        pattern: InkPattern,
   254	        anchor: Bounds,
   255	        ids: IdGenerator,
   256	        startTime: Long = 0L,
   257	        groupId: String = ids.next(),

   262	        val src = pattern.outputInk.bounds()
   263	        // Sin zona elástica, lo que se mide y se centra en el gesto es el núcleo (la salida entera si no tiene).
   264	        val core = PatternCore.bounds(pattern)
   265	        val eps = 0.5f
   266	        val zone = pattern.stretch?.takeIf { elastic }?.sanitized()

   270	        val scaleW = if (ref.width > eps) anchor.width / ref.width else Float.NaN
   271	        val s = when (pattern.sizing) {
   272	            OutputSizing.MATCH_HEIGHT -> scaleH.orElse(scaleW)
   273	            OutputSizing.MATCH_WIDTH -> scaleW.orElse(scaleH)
   274	            OutputSizing.FIT_GESTURE -> minOf(scaleW.orElse(scaleH), scaleH.orElse(scaleW))
   275	            OutputSizing.FIXED -> listOfNotNull(

   277	                if (src.height > eps) pattern.fixedHeight / src.height else null,

   285	                    anchor.maxSide / ref.maxSide.coerceAtLeast(eps)
   286	                else -> raw
   287	            }

   290	            stretched(pattern.outputInk, src, anchor, s, zone)
   291	        } else {
   292	            val t = Affine.translate(-core.centerX, -core.centerY)
   293	                .then(Affine(a = s, d = s))
   294	                .then(Affine.translate(anchor.centerX, anchor.centerY))
   295	            pattern.outputInk.map { it.transformed(t, scaleWidth = false) }
   296	        }
   297	        return placed.map { stroke ->
   298	            stroke.adopting(style).let { if (presetColor != null) it.copy(color = presetColor) else it }.copy(
   299	                id = ids.next(),
   300	                startTime = startTime,
   301	                groupId = groupId,
   302	                patternId = pattern.id,
   303	            )
   304	        }
   305	    }
   306	
   307	    /**
   308	     * Escala uniforme [s] salvo en la zona elástica, que absorbe la diferencia hasta el borde del gesto:
   309	     * la salida empieza en el borde izquierdo (o superior) de [anchor] y termina en el opuesto, salvo que la
   310	     * zona ya no pueda encogerse más ([Stretch.minFactor]). En el otro eje se centra como siempre.
   311	     */
   312	    private fun stretched(ink: List<InkStroke>, src: Bounds, anchor: Bounds, s: Float, zone: Stretch): List<InkStroke> {
   313	        val horizontal = zone.axis == Axis.HORIZONTAL
   314	        val start = if (horizontal) src.left else src.top
   315	        val natural = (if (horizontal) src.width else src.height) * s
   316	        val target = if (horizontal) anchor.width else anchor.height
   317	        val za = zone.from * natural
   318	        val zb = zone.to * natural
   319	        val len = zb - za
   320	        val newLen = maxOf(len + target - natural, zone.minFactor * len)
   321	        val k = if (len > 0f) newLen / len else 1f
   322	        val delta = newLen - len
   323	        fun along(v: Float): Float {
   324	            val u = (v - start) * s
   325	            return when {
   326	                u <= za -> u
   327	                u >= zb -> u + delta
   328	                else -> za + (u - za) * k
   329	            }
   330	        }
   331	        val origin = if (horizontal) anchor.left else anchor.top
   332	        val crossFrom = if (horizontal) src.centerY else src.centerX
   333	        val crossTo = if (horizontal) anchor.centerY else anchor.centerX
   334	        return ink.map { stroke ->
   335	            stroke.copy(points = stroke.points.map { p ->
   336	                if (horizontal) p.copy(x = origin + along(p.x), y = (p.y - crossFrom) * s + crossTo)
   337	                else p.copy(x = (p.x - crossFrom) * s + crossTo, y = origin + along(p.y))
   338	            })
   339	        }
   340	    }
   341	
   342	    /** Zona saneada (finita, dentro de [0, 1] y con largo); null si no sirve. */
   343	    private fun Stretch.sanitized(): Stretch? {
   344	        if (!from.isFinite() || !to.isFinite() || !minFactor.isFinite()) return null
   345	        val a = from.coerceIn(0f, 1f)
   346	        val b = to.coerceIn(0f, 1f)
   347	        if (b - a < 0.01f) return null
   348	        return copy(from = a, to = b, minFactor = minFactor.coerceIn(0f, 1f))
   349	    }

   351	    internal fun InkStroke.adopting(style: InkStyle?): InkStroke {
   352	        // El resaltador no se hereda: la salida es tinta.
   353	        if (style == null || style.brush == Brush.HIGHLIGHTER || color != DEFAULT_INK_COLOR) return this
   354	        return copy(
   355	            color = style.color,
   356	            width = (width * style.width / DEFAULT_INK_WIDTH).coerceIn(0.5f, 60f),
   357	            brush = if (brush == Brush.PEN) style.brush else brush,
   358	        )
   359	    }

   361	    private fun Float.orElse(other: Float) = if (isNaN()) other else this
   362	}

    14	    /**
    15	     * @param gesture trazos del gesto en orden.
    16	     * @param dy corrimiento vertical del alineado automático (0 = donde se dibujó).
    17	     * @return trazos de salida (✓ y barra) o null si el gesto no tiene forma de raíz (se usa la plantilla).
    18	     */
    19	    fun fit(
    20	        pattern: InkPattern,
    21	        gesture: List<InkStroke>,
    22	        ids: IdGenerator,
    23	        startTime: Long,
    24	        style: InkStyle?,
    25	        dy: Float = 0f,
    26	    ): List<InkStroke>? {
    27	        if (pattern.outputInk.size < 2 || gesture.isEmpty()) return null
    28	        val pts = gesture.flatMap { it.points }
    29	        if (pts.size < 4) return null
    30	        val b = Bounds.of(pts)
    31	        val h = b.height
    32	        if (h < 1f || b.width < 1f) return null
    33	        // Punta del ✓: el primer punto que llega arriba después de haber bajado.
    34	        val top = b.top
    35	        var wentDown = false
    36	        var peak = -1
    37	        for (i in pts.indices) {
    38	            val y = pts[i].y
    39	            if (y > top + 0.4f * h) wentDown = true
    40	            if (wentDown && y <= top + 0.12f * h) { peak = i; break }
    41	        }
    42	        if (peak < 2) return null
    43	        val check = smooth(pts.subList(0, peak + 1)).map { it.copy(y = it.y + dy) }
    44	        val tip = check.last()
    45	        val end = max(b.right, tip.x + 0.3f * h)
    46	        val bar = (0..BAR_POINTS).map { k ->
    47	            val u = k / BAR_POINTS.toFloat()
    48	            InkPoint(tip.x + (end - tip.x) * u, tip.y, pressure = taper(u), timeMs = tip.timeMs + k * 8L)
    49	        }
    50	        val checkTpl = pattern.outputInk.first()
    51	        val barTpl = pattern.outputInk.last()
    52	        val group = ids.next()
    53	        return listOf(checkTpl.copy(points = check), barTpl.copy(points = bar)).map { s ->
    54	            s.adopting(style).copy(id = ids.next(), startTime = startTime, groupId = group, patternId = pattern.id)
    55	        }
    56	    }

    58	    /** Media móvil de 3 puntos (los extremos quedan fijos); afina la presión en las puntas. */
    59	    private fun smooth(src: List<InkPoint>): List<InkPoint> {
    60	        val n = src.size
    61	        return src.mapIndexed { i, p ->
    62	            val u = if (n > 1) i / (n - 1f) else 0f
    63	            val pr = min(p.pressure.coerceIn(0.2f, 1f), taper(u))
    64	            if (i == 0 || i == n - 1) p.copy(pressure = pr)
    65	            else p.copy(
    66	                x = (src[i - 1].x + p.x + src[i + 1].x) / 3f,
    67	                y = (src[i - 1].y + p.y + src[i + 1].y) / 3f,
    68	                pressure = pr,
    69	            )
    70	        }
    71	    }

    73	    private fun taper(u: Float): Float = (0.45f + 0.55f * min(1f, min(u, 1f - u) * 6f)).coerceIn(0.3f, 1f)

   242	    private fun closeAfterHand(handTop: Float, brush: Brush? = null): Pair<Bounds, Bounds> {
   243	        val open = preset("parentesis-abre")
   244	        val close = preset("parentesis-cierra")
   245	        val opener = InkReplacement.place(open, Bounds(100f, 100f, 140f, 300f), IdGenerator.sequential())
   246	        val hand = listOf(300f, 500f, 700f).mapIndexed { i, x ->
   247	            val s = InkStroke("h$i", listOf(InkPoint(x, handTop), InkPoint(x + 60f, handTop + 100f)))
   248	            if (brush == null) s else s.copy(brush = brush)
   249	        }
   250	        val gesture = InkStroke("g", listOf(InkPoint(900f, 160f), InkPoint(920f, 200f), InkPoint(900f, 240f)))
   251	        val c = InkController(PageState(opener + hand + gesture), History(), { listOf(open, close) }, IdGenerator.sequential())
   252	        return opener.bounds() to c.convertWith(close, listOf("g"))!!.bounds()
   253	    }
